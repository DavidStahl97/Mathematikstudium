\section*{Aufgabe 3.4}

\begin{enumerate}[label=\alph*)]

    \item
          \[
              \frac{\partial f(r)}{\partial x} = \frac{\partial f\bigl(\sqrt{x^2+y^2+z^2}\bigr)}{\partial x} = 2x \cdot \frac{1}{2r} \cdot f'(r) = f'(r) \cdot \frac{x}{r}
          \]

          \[
              \frac{\partial f(r)}{\partial y} = f'(r) \cdot \frac{y}{r}
          \]

          \[
              \frac{\partial f(r)}{\partial z} = f'(r) \cdot \frac{z}{r}
          \]

          \[
              \Rightarrow \operatorname{grad} f(r) = f'(r) \cdot \frac{(x,y,z)}{r}
          \]

    \item
          \[
              \frac{\partial^2 f}{\partial x^2} = \frac{\partial \, f'(r) \cdot \frac{x}{r}}{\partial x} = \frac{r \left( \frac{x^2}{r} f''(r) + f'(r) \right) - f'(r) \cdot \frac{x^2}{r}}{r}
          \]

          \[
              = \frac{x^2 f''(r)}{r^2} + \frac{f'(r)}{r} - \frac{f'(r) \cdot x^2}{r^3}
          \]

          \begin{align*}
              \Delta f(r) & = \frac{f''(r)}{r^2} (x^2+y^2+z^2) + 3 \frac{f'(r)}{r} - \frac{f'(r)}{r^3} (x^2+y^2+z^2) \\
                          & = f''(r) + 3 \frac{f'(r)}{r} - \frac{f'(r)}{r}                                          \\
                          & = f''(r) + 2 \frac{f'(r)}{r}
          \end{align*}

    \item
          \[
              f : (0, \infty) \to \mathbb{R}, \quad x \mapsto \frac{1}{x}
          \]

          \[
              f'(x) = -\frac{1}{x^2}
          \]

          \[
              f''(x) = \frac{2}{x^3}
          \]

          Nach b) folgt
          \begin{align*}
              \Delta\left(\frac{1}{r}\right) = \Delta f(r) & = f''(r) + \frac{2}{r} f'(r)                       \\
                                                           & = \frac{2}{r^3} + \frac{2}{r} \left(-\frac{1}{r^2}\right) \\
                                                           & = 0
          \end{align*}

\end{enumerate}
