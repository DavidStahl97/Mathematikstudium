\section*{Aufgabe 1}
\begin{enumerate}[label=\alph*)]
  \item Wegen SinCos-2 gilt $\sin(0) = -\sin(0)$.
        Daraus folgt, dass $\sin(0)$ das neutrale Element $0$ ist.

  \item Aus Satz 16.3.11 und Kettenregel folgt:
        \[
            \left(\frac{1}{\sin(x)}\right)' = -\frac{\cos(x)}{\sin(x)^2}
        \]
        (für alle $x \in \mathbb{R} \setminus \{0\}$)
\end{enumerate}
