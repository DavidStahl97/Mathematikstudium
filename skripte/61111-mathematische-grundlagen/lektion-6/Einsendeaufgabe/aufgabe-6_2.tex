\section*{Aufgabe 6.2}

\begin{enumerate}[label=\alph*)]

\item
\[
  \lim_{x \to 0} \frac{\cos(x) + 3x - 1}{2x}
  = \lim_{x \to 0} \frac{-\sin(x) + 3}{2}
  = \frac{3}{2}
\]

\item
\begin{align*}
  \lim_{x \to \frac{\pi}{2}} \left(x - \frac{\pi}{2}\right) \tan(x)
  &= \lim_{x \to \frac{\pi}{2}} \frac{\left(x - \frac{\pi}{2}\right) \sin(x)}{\cos(x)} \\
  &= \lim_{x \to \frac{\pi}{2}} \frac{\sin(x) - \left(x - \frac{\pi}{2}\right) \cos(x)}{-\sin(x)}
  = -1
\end{align*}

\end{enumerate}
