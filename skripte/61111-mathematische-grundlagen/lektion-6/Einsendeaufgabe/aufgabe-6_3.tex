\section*{Aufgabe 6.3}

\begin{enumerate}[label=\alph*)]

\item
\[
  f(x) = \sqrt[3]{x}
\]
\begin{align*}
  f'(x) &= \frac{1}{3} \cdot \left(\frac{1}{\sqrt[3]{x}}\right)^{2} \\[6pt]
  f''(x) &= \frac{1}{3} \cdot \left(-\frac{2}{3}\right) \cdot \left(\frac{1}{\sqrt[3]{x}}\right)^{5}
           = -\frac{2}{9} \cdot \left(\frac{1}{\sqrt[3]{x}}\right)^{5} \\[6pt]
  f'''(x) &= -\frac{2}{9} \cdot \left(-\frac{5}{3}\right) \cdot \left(\frac{1}{\sqrt[3]{x}}\right)^{8}
            = \frac{10}{27} \cdot \left(\frac{1}{\sqrt[3]{x}}\right)^{8}
\end{align*}

Entwicklungspunkt: $27$
\begin{align*}
  f(27)   &= 3 \\
  f'(27)  &= \frac{1}{3^3} \\
  f''(27) &= -\frac{2}{3^7} \\
  f'''(27) &= \frac{10}{3^{11}}
\end{align*}

\begin{align*}
  P_{2,27}(30)
  &= 3 + \frac{1}{3^3} \cdot (30-27) - \frac{1}{3^7} \cdot (30-27)^{2} \\[4pt]
  &= 3 + \frac{1}{3^2} - \frac{1}{3^5}
\end{align*}

\[
  \lvert R_{2,27}(30) \rvert
  = \frac{f'''(\xi)}{3!} \cdot (30-27)^{3}
  < \frac{\dfrac{10}{3^{11}}}{3!} \cdot 3^{3}
  = \frac{5}{3^9}
\]

\item
\[
  f(x) = \ln(x)
\]
\begin{align*}
  f'(x)   &= \frac{1}{x} \\[4pt]
  f''(x)  &= -\frac{1}{x^2} \\[4pt]
  f'''(x) &= \frac{2}{x^3}
\end{align*}

Entwicklungspunkt: $1$
\begin{align*}
  f(1)    &= 0 \\
  f'(1)   &= 1 \\
  f''(1)  &= -1 \\
  f'''(1) &= 2
\end{align*}

\begin{align*}
  P_{2,1}\!\left(\frac{5}{6}\right)
  &= -\frac{1}{6} - \frac{1}{2} \cdot \left(-\frac{1}{6}\right)^{2} \\[4pt]
  &= -\frac{1}{6}\left(1 + \frac{1}{12}\right)
   = -\frac{1}{6} \cdot \frac{13}{12}
   = -\frac{13}{72}
\end{align*}

\[
  \left\lvert R_{2,1}\!\left(\frac{5}{6}\right) \right\rvert
  = \left\lvert \frac{f'''(\xi)}{3!} \cdot \left(-\frac{1}{6}\right)^{3} \right\rvert
  < \left\lvert \frac{2}{3!} \cdot \left(-\frac{1}{6}\right)^{3} \right\rvert
  = \left\lvert \frac{1}{3} \cdot \left(-\frac{1}{216}\right) \right\rvert
  = \frac{1}{648}
\]

\end{enumerate}
